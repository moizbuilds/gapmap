# GapMap Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build GapMap — paste a resume + JD, get per-requirement match/partial/missing verdicts with programmatically verified resume quotes, a code-computed readiness score, gap-closing suggestions, a shareable score card, localStorage history with comparison, and an eval harness that decides the judge model.

```ts
// lib/normalize.ts
// One normalization used by BOTH the quote verifier (lib/pipeline/verify.ts)
// and the UI evidence highlighter. Sharing it means "verified" and
// "highlightable" are the same fact — they can never drift apart.
//
// CONCEPT: normalization = reducing two texts to a common form (lowercase,
// straight quotes, single spaces) so cosmetic differences don't defeat a
// substring check. NFKC is Unicode's "compatibility" normal form — it folds
// lookalike characters (e.g. the ﬁ ligature) into their plain equivalents.

const CHAR_MAP: Record<string, string> = {
  "‘": "'", "’": "'",   // ‘ ’
  "“": '"', "”": '"',   // “ ”
  "–": "-", "—": "-",   // – —
  "…": "...",                 // …
  " ": " ",                   // non-breaking space
};

export interface NormalizedText { text: string; offsets: number[] }

// CONCEPT: offset mapping — while normalizing we record, for every output
// char, which input index it came from. That lets us find a quote in the
// normalized text but highlight it in the user's ORIGINAL resume.
// WHY per-char NFKC (not whole-string): whole-string NFKC can change length
// before we build the map, which would corrupt every offset after it.
export function normalizeWithMap(input: string): NormalizedText {
  const chars: string[] = [];
  const offsets: number[] = [];
  let lastWasSpace = true; // starting true trims leading whitespace

  for (let i = 0; i < input.length; i++) {
    const mapped = CHAR_MAP[input[i]] ?? input[i];
    for (const ch of mapped.normalize("NFKC").toLowerCase()) {
      if (/\s/.test(ch)) {
        if (lastWasSpace) continue;      // collapse whitespace runs
        chars.push(" ");
        offsets.push(i);
        lastWasSpace = true;
      } else {
        chars.push(ch);
        offsets.push(i);
        lastWasSpace = false;
      }
    }
  }
  while (chars.length && chars[chars.length - 1] === " ") { chars.pop(); offsets.pop(); } // trim trailing
  return { text: chars.join(""), offsets };
}

export function normalize(input: string): string {
  return normalizeWithMap(input).text;
}

export function findQuoteInOriginal(
  original: string,
  quote: string,
): { start: number; end: number } | null {
  const needle = normalize(quote);
  if (needle.length === 0) return null;
  const { text, offsets } = normalizeWithMap(original);
  const idx = text.indexOf(needle);
  if (idx === -1) return null;
  const start = offsets[idx];
  const end = offsets[idx + needle.length - 1] + 1; // +1: end is exclusive
  return { start, end };
}
```

```ts
// lib/validate.ts
// Server-side validation of the analyze request. The client mirrors these
// checks for UX, but the server is the one that counts — anything in a
// request body is attacker-controlled (trust boundary rule).

import { RESUME_MAX_CHARS, JD_MAX_CHARS } from "@/lib/constants";
import type { AnalyzeRequest, AnalyzeError } from "@/types/analysis";

type Result = { ok: true; data: AnalyzeRequest } | { ok: false; error: AnalyzeError; status: number };

export function validateAnalyzeRequest(body: unknown): Result {
  if (typeof body !== "object" || body === null || Array.isArray(body)) {
    return { ok: false, error: { error: "INPUT_EMPTY", field: "resumeText" }, status: 400 };
  }
  const { resumeText, jdText } = body as Record<string, unknown>;

  for (const [field, value, max] of [
    ["resumeText", resumeText, RESUME_MAX_CHARS],
    ["jdText", jdText, JD_MAX_CHARS],
  ] as const) {
    if (typeof value !== "string" || value.trim().length === 0) {
      return { ok: false, error: { error: "INPUT_EMPTY", field }, status: 400 };
    }
    if (value.length > max) {
      return { ok: false, error: { error: "INPUT_TOO_LONG", field, max }, status: 400 };
    }
  }
  return { ok: true, data: { resumeText: resumeText as string, jdText: jdText as string } };
}
```

```ts
// lib/pipeline/score.ts
// The readiness score is computed HERE, in plain arithmetic — never by the
// LLM. That makes it deterministic, explainable row by row in the UI, and
// impossible to inflate. (Spec §7 fixes this formula.)

import type { Requirement, Judgment, ScoreBreakdown } from "@/types/analysis";

const VALUE: Record<Judgment["verdict"], number> = { match: 1, partial: 0.5, missing: 0 };
const WEIGHT: Record<Requirement["priority"], number> = { must_have: 2, nice_to_have: 1 };

function band(score: number): ScoreBreakdown["band"] {
  if (score >= 85) return "strong_fit";
  if (score >= 65) return "competitive";
  if (score >= 40) return "stretch";
  return "not_yet";
}

export function computeScore(requirements: Requirement[], judgments: Judgment[]): ScoreBreakdown {
  if (requirements.length === 0) throw new Error("computeScore: no requirements");
  const byId = new Map(judgments.map((j) => [j.requirementId, j]));

  const counts = { match: 0, partial: 0, missing: 0 };
  let earned = 0, possible = 0;
  const rows = requirements.map((r) => {
    // A requirement without a judgment can't happen after judge.ts backfills,
    // but scoring treats it as missing anyway — fail toward honesty.
    const verdict = byId.get(r.id)?.verdict ?? "missing";
    counts[verdict]++;
    const weight = WEIGHT[r.priority];
    const value = VALUE[verdict];
    earned += weight * value;
    possible += weight;
    return { requirementId: r.id, weight, value, contribution: weight * value };
  });

  const score = Math.round((100 * earned) / possible);
  return { score, band: band(score), counts, rows };
}
```

```ts
// lib/pipeline/verify.ts
// The honesty net. Every evidence quote the judge LLM produced is checked
// against the actual resume text; quotes that aren't really there are
// dropped, and a match/partial verdict that loses ALL its evidence is
// downgraded to missing and flagged. After this stage, fabricated evidence
// cannot reach the user — structurally, not by prompt-engineering hope.

import { findQuoteInOriginal } from "@/lib/normalize";
import type { RawJudgment, Judgment, EvidenceQuote } from "@/types/analysis";

const MAX_QUOTE_CHARS = 300;
const MAX_QUOTES = 3;

export const EVIDENCE_REJECTED_SUGGESTION =
  "The analysis claimed support it couldn't actually quote from your resume, so this requirement is treated as missing. If you do have this experience, state it explicitly in a bullet.";

export function verifyJudgments(resumeText: string, raw: RawJudgment[]): Judgment[] {
  return raw.map((rj) => {
    const evidence: EvidenceQuote[] = [];
    let dropped = 0;

    for (const quote of rj.evidenceQuotes.slice(0, MAX_QUOTES)) {
      const hit = quote.length <= MAX_QUOTE_CHARS ? findQuoteInOriginal(resumeText, quote) : null;
      if (hit) {
        // Store the ORIGINAL resume slice + offsets as first-class fields so
        // the UI highlights real text instead of re-searching (and possibly
        // disagreeing with) the verifier.
        evidence.push({ quote: resumeText.slice(hit.start, hit.end), startOffset: hit.start, endOffset: hit.end });
      } else {
        dropped++;
      }
    }
    dropped += Math.max(0, rj.evidenceQuotes.length - MAX_QUOTES);

    const claimedSupport = rj.verdict !== "missing";
    const rejected = claimedSupport && evidence.length === 0 && rj.evidenceQuotes.length > 0;
    // A match/partial with NO quotes at all is also unsupported — same downgrade.
    const unsupported = claimedSupport && evidence.length === 0;

    return {
      requirementId: rj.requirementId,
      verdict: unsupported ? "missing" : rj.verdict,
      evidence: rj.verdict === "missing" ? [] : evidence,
      reasoning: rj.reasoning,
      suggestion: rejected ? EVIDENCE_REJECTED_SUGGESTION : rj.suggestion,
      rewrittenBullet: unsupported ? null : rj.rewrittenBullet,
      evidenceRejected: rejected,
      judgeSkipped: false,
      droppedQuoteCount: dropped,
    };
  });
}
```

```ts
// lib/pipeline/extract.ts
// Pipeline stage 1: job description → structured requirements checklist.
//
// CONCEPT: forced tool use — we give the model exactly one "tool" whose input
// schema IS our output type, and force it to call that tool. The response
// arrives as schema-shaped JSON instead of free text, so there's no
// markdown-fence parsing. We still scan response.content for the tool_use
// block instead of assuming content[0] — a thinking block may come first
// (this exact assumption broke a previous build).

import type Anthropic from "@anthropic-ai/sdk";
import { EXTRACT_MODEL, EXTRACT_MAX_TOKENS, MAX_REQUIREMENTS } from "@/lib/constants";
import type { ExtractionResult, Requirement, Category, Priority } from "@/types/analysis";

const CATEGORIES: Category[] = ["hard_skill", "soft_skill", "experience", "education", "domain", "other"];
const PRIORITIES: Priority[] = ["must_have", "nice_to_have"];

const TOOL: Anthropic.Tool = {
  name: "record_requirements",
  description: "Record the structured hiring requirements extracted from the job description.",
  input_schema: {
    type: "object",
    properties: {
      jobTitle: { type: ["string", "null"] },
      company: { type: ["string", "null"] },
      requirements: {
        type: "array",
        items: {
          type: "object",
          properties: {
            text: { type: "string" },
            category: { type: "string", enum: CATEGORIES },
            priority: { type: "string", enum: PRIORITIES },
          },
          required: ["text", "category", "priority"],
        },
      },
    },
    required: ["jobTitle", "company", "requirements"],
  },
};

const prompt = (jdText: string) => `You are extracting hiring requirements from a job description, the way a hiring manager would build a screening checklist.

<job_description>
${jdText}
</job_description>

Extract every distinct requirement a hiring manager would actually screen for. Rules:
- One requirement per distinct qualification. Split bundled lists ("React, TypeScript, and GraphQL") into separate requirements when they'd be screened separately; keep genuinely single skills together.
- priority = "must_have" when stated as required/essential/minimum or clearly core to the role; "nice_to_have" when marked preferred/plus/bonus or clearly peripheral.
- category = one of: hard_skill, soft_skill, experience, education, domain, other.
- Ignore boilerplate (EEO statements, benefits, "fast-paced environment" filler).
- Order by importance, most important first. Maximum ${MAX_REQUIREMENTS} requirements.
- Also extract jobTitle and company when present (null when absent).`;

export async function extractRequirements(client: Anthropic, jdText: string): Promise<ExtractionResult> {
  const response = await client.messages.create({
    model: EXTRACT_MODEL,
    max_tokens: EXTRACT_MAX_TOKENS,
    tools: [TOOL],
    tool_choice: { type: "tool", name: "record_requirements" },
    messages: [{ role: "user", content: prompt(jdText) }],
  });

  const block = response.content.find((b) => b.type === "tool_use");
  if (!block || block.type !== "tool_use") throw new Error("extract: no tool_use block in response");
  const input = block.input as {
    jobTitle?: unknown; company?: unknown;
    requirements?: { text?: unknown; category?: unknown; priority?: unknown }[];
  };

  // Defensive narrowing: forced tool use is schema-guided, not schema-
  // guaranteed. Bad enum values are clamped conservatively (nice_to_have
  // weighs LESS, so a clamp can't inflate the score).
  const cleaned: Requirement[] = (Array.isArray(input.requirements) ? input.requirements : [])
    .filter((r) => typeof r.text === "string" && r.text.trim().length > 0)
    .map((r, i) => ({
      id: `req_${i + 1}`,
      text: (r.text as string).trim(),
      category: CATEGORIES.includes(r.category as Category) ? (r.category as Category) : "other",
      priority: PRIORITIES.includes(r.priority as Priority) ? (r.priority as Priority) : "nice_to_have",
    }));

  return {
    jobTitle: typeof input.jobTitle === "string" ? input.jobTitle : null,
    company: typeof input.company === "string" ? input.company : null,
    requirements: cleaned.slice(0, MAX_REQUIREMENTS),
    requirementsTruncated: cleaned.length > MAX_REQUIREMENTS,
  };
}
```

```ts
// lib/pipeline/judge.ts
// Pipeline stage 2: the screener. Given the requirements checklist and the
// full resume, produce a verdict + verbatim quotes per requirement. Output
// here is RAW — stage 3 (verify.ts) checks every quote before anything is
// shown to a user.

import type Anthropic from "@anthropic-ai/sdk";
import { JUDGE_MAX_TOKENS } from "@/lib/constants";
import type { Requirement, RawJudgment, VerdictLabel } from "@/types/analysis";

const VERDICTS: VerdictLabel[] = ["match", "partial", "missing"];

const TOOL: Anthropic.Tool = {
  name: "record_judgments",
  description: "Record the verdict for every requirement id.",
  input_schema: {
    type: "object",
    properties: {
      judgments: {
        type: "array",
        items: {
          type: "object",
          properties: {
            requirementId: { type: "string" },
            verdict: { type: "string", enum: VERDICTS },
            evidence: { type: "array", items: { type: "string" }, maxItems: 3 },
            reasoning: { type: "string" },
            suggestion: { type: ["string", "null"] },
            rewrittenBullet: { type: ["string", "null"] },
          },
          required: ["requirementId", "verdict", "evidence", "reasoning", "suggestion", "rewrittenBullet"],
        },
      },
    },
    required: ["judgments"],
  },
};

const prompt = (resumeText: string, requirements: Requirement[]) => `You are a rigorous, honest hiring screener judging a candidate's resume against a list of job requirements. Your judgments will be shown to the candidate with your quoted evidence, and every quote will be programmatically checked against the resume — a quote that is not a verbatim substring of the resume will be discarded and your verdict downgraded.

<resume>
${resumeText}
</resume>

<requirements>
${JSON.stringify(requirements.map(({ id, text, category, priority }) => ({ id, text, category, priority })), null, 2)}
</requirements>

For EVERY requirement id, return a judgment:

verdict rules:
- "match": the resume clearly demonstrates this requirement with concrete evidence (work done, results, duration).
- "partial": related or adjacent evidence exists but doesn't fully satisfy the requirement — OR the skill is merely listed (e.g. in a skills section) with no demonstrated use. Keyword mention alone is never a "match".
- "missing": no supporting evidence in the resume. When in doubt between partial and missing, and you cannot quote supporting text, choose missing.

evidence rules (critical):
- 1-3 quotes per judgment, each an EXACT verbatim substring copied character-for-character from the resume, max 300 characters each.
- Never paraphrase, merge, truncate mid-word, or "fix" the resume's wording inside a quote.
- "missing" verdicts have an empty evidence array.

suggestion rules:
- match: null, or one short optional strengthening tip.
- partial: a concrete, specific improvement — either a rewrite of one of the QUOTED bullets (grounded strictly in what the quote already claims — add specificity/metrics, never new claims), or a specific addition. If a rewrite, also return rewrittenBullet with the copy-ready line; otherwise rewrittenBullet is null.
- missing: a concrete action to close the gap (a specific project to build, a certification, a skill to learn and how to evidence it). NEVER write resume text claiming experience the candidate does not have. rewrittenBullet is always null for missing.

reasoning: one sentence explaining the verdict, plain language.`;

export async function judgeRequirements(
  client: Anthropic,
  model: string, // parameter, not constant: the eval compares Haiku vs Sonnet here
  resumeText: string,
  requirements: Requirement[],
): Promise<{ raw: RawJudgment[]; skippedIds: string[] }> {
  const response = await client.messages.create({
    model,
    max_tokens: JUDGE_MAX_TOKENS,
    tools: [TOOL],
    tool_choice: { type: "tool", name: "record_judgments" },
    messages: [{ role: "user", content: prompt(resumeText, requirements) }],
  });

  const block = response.content.find((b) => b.type === "tool_use");
  if (!block || block.type !== "tool_use") throw new Error("judge: no tool_use block in response");
  const items = (block.input as { judgments?: unknown[] }).judgments;
  if (!Array.isArray(items)) throw new Error("judge: judgments is not an array");

  // Index the model's output by id: keep the FIRST judgment per id, drop ids
  // we never asked about (the model can't invent requirements).
  const byId = new Map<string, RawJudgment>();
  for (const it of items) {
    const j = it as Record<string, unknown>;
    const id = typeof j.requirementId === "string" ? j.requirementId : "";
    if (!id || byId.has(id)) continue;
    const verdict = VERDICTS.includes(j.verdict as VerdictLabel) ? (j.verdict as VerdictLabel) : "missing";
    byId.set(id, {
      requirementId: id,
      verdict,
      // missing may not carry evidence; rewrittenBullet exists ONLY for partial
      evidenceQuotes: verdict === "missing" ? [] :
        (Array.isArray(j.evidence) ? j.evidence.filter((q): q is string => typeof q === "string") : []),
      reasoning: typeof j.reasoning === "string" ? j.reasoning : "",
      suggestion: typeof j.suggestion === "string" ? j.suggestion : null,
      rewrittenBullet: verdict === "partial" && typeof j.rewrittenBullet === "string" ? j.rewrittenBullet : null,
    });
  }

  // Backfill any requirement the judge skipped as an honest "missing",
  // reported separately so pipeline.ts can flag it (judgeSkipped) in the UI.
  const skippedIds: string[] = [];
  const raw = requirements.map((r) => {
    const found = byId.get(r.id);
    if (found) return found;
    skippedIds.push(r.id);
    return { requirementId: r.id, verdict: "missing" as const, evidenceQuotes: [],
             reasoning: "The screener did not return a judgment for this requirement.",
             suggestion: null, rewrittenBullet: null };
  });
  return { raw, skippedIds };
}
```

```ts
// lib/pipeline/pipeline.ts
// The conductor: extract → judge → verify → score, in order, or a typed
// failure. Takes the Anthropic client and judge model as PARAMETERS so unit
// tests and the eval harness drive the exact same code path as production.

import type Anthropic from "@anthropic-ai/sdk";
import { extractRequirements } from "./extract";
import { judgeRequirements } from "./judge";
import { verifyJudgments } from "./verify";
import { computeScore } from "./score";
import type { AnalyzeRequest, AnalysisResult } from "@/types/analysis";

export class PipelineError extends Error {
  constructor(public code: "NO_REQUIREMENTS", message: string) {
    super(message);
    this.name = "PipelineError";
  }
}

export async function runAnalysis(
  client: Anthropic,
  judgeModel: string,
  input: AnalyzeRequest,
): Promise<AnalysisResult> {
  const extraction = await extractRequirements(client, input.jdText);
  if (extraction.requirements.length === 0) {
    throw new PipelineError("NO_REQUIREMENTS", "no requirements extracted — input likely not a JD");
  }

  const { raw, skippedIds } = await judgeRequirements(client, judgeModel, input.resumeText, extraction.requirements);
  const skipped = new Set(skippedIds);
  const judgments = verifyJudgments(input.resumeText, raw)
    .map((j) => (skipped.has(j.requirementId) ? { ...j, judgeSkipped: true } : j));

  return {
    id: crypto.randomUUID(),
    createdAt: new Date().toISOString(), // stamped server-side (time rule)
    extraction,
    judgments,
    scoreBreakdown: computeScore(extraction.requirements, judgments),
    judgeModel,
  };
}
```

```ts
// lib/ratelimit.ts
// Rate limiting that survives serverless. An in-memory counter dies because
// every request may run on a fresh instance — so counters live in Upstash
// Redis (they hold hashed IPs only, never resume data).
//
// CONCEPT: sliding window — "10 per hour" measured over the trailing 60
// minutes, not a counter that resets on the hour (which would allow 20
// back-to-back around the boundary).

import { Ratelimit } from "@upstash/ratelimit";
import { Redis } from "@upstash/redis";
import { createHash } from "node:crypto";
import { RATE_LIMIT_PER_HOUR } from "@/lib/constants";
import { ServiceUnavailableError } from "@/lib/errors";

// TRUST BOUNDARY: x-forwarded-for is client-forgeable (a rotating value
// fully bypassed a rate limit in a previous build). On Vercel, the platform
// strips inbound x-vercel-forwarded-for and sets it itself — that's the only
// header we trust. Anything else → treat as local dev.
export function clientIpFrom(headers: Headers): string {
  const vercel = headers.get("x-vercel-forwarded-for");
  if (vercel) return vercel.split(",")[0].trim();
  return "127.0.0.1";
}

let limiter: Ratelimit | null = null;
function getLimiter(): Ratelimit | null {
  const { UPSTASH_REDIS_REST_URL: url, UPSTASH_REDIS_REST_TOKEN: token } = process.env;
  if (!url || !token) {
    if (process.env.NODE_ENV === "development") {
      console.warn("[gapmap] Upstash env missing — rate limiting DISABLED (dev only)");
      return null; // dev convenience; production fails closed below
    }
    throw new ServiceUnavailableError("Upstash env missing — failing closed");
  }
  if (!limiter) {
    limiter = new Ratelimit({
      redis: new Redis({ url, token }),
      limiter: Ratelimit.slidingWindow(RATE_LIMIT_PER_HOUR, "1 h"),
      prefix: "gapmap:rl",
    });
  }
  return limiter;
}

export async function checkRateLimit(
  req: Request,
): Promise<{ allowed: true } | { allowed: false; retryAfterSeconds: number }> {
  const rl = getLimiter();
  if (!rl) return { allowed: true }; // dev without Upstash

  // Hash the IP before using it as a Redis key — we don't need to store the
  // raw address to count it.
  const ipHash = createHash("sha256").update(clientIpFrom(req.headers)).digest("hex");
  const { success, reset } = await rl.limit(ipHash);
  if (success) return { allowed: true };
  return { allowed: false, retryAfterSeconds: Math.max(1, Math.ceil((reset - Date.now()) / 1000)) };
}
```

```ts
// app/api/analyze/route.ts
// The single API endpoint. Order matters and mirrors spec §9:
// fail-closed env check → parse → validate → rate limit → pipeline.
// Every failure returns a typed AnalyzeError — the UI switches on `error`,
// never on prose.

import { NextResponse } from "next/server";
import { validateAnalyzeRequest } from "@/lib/validate";
import { checkRateLimit } from "@/lib/ratelimit";
import { getAnthropicClient } from "@/lib/anthropic";
import { runAnalysis, PipelineError } from "@/lib/pipeline/pipeline";
import { JUDGE_MODEL, RESUME_MAX_CHARS } from "@/lib/constants";
import { ServiceUnavailableError } from "@/lib/errors";
import type { AnalyzeError } from "@/types/analysis";

export const runtime = "nodejs";
export const maxDuration = 60; // two sequential LLM calls need more than the 10s default

const err = (e: AnalyzeError, status: number) => NextResponse.json(e, { status });

export async function POST(req: Request) {
  try {
    // Reject huge bodies before JSON.parse allocates for them.
    const contentLength = Number(req.headers.get("content-length") ?? 0);
    if (contentLength > 200_000) return err({ error: "INPUT_TOO_LONG", field: "resumeText", max: RESUME_MAX_CHARS }, 400);

    let body: unknown = {};
    try { body = await req.json(); } catch { /* falls through to validation → INPUT_EMPTY */ }
    const v = validateAnalyzeRequest(body);
    if (!v.ok) return err(v.error, v.status);

    const rl = await checkRateLimit(req);
    if (!rl.allowed) return err({ error: "RATE_LIMITED", retryAfterSeconds: rl.retryAfterSeconds }, 429);

    const result = await runAnalysis(getAnthropicClient(), JUDGE_MODEL, v.data);
    return NextResponse.json(result);
  } catch (e) {
    if (e instanceof ServiceUnavailableError) return err({ error: "SERVICE_UNAVAILABLE" }, 503);
    if (e instanceof PipelineError) return err({ error: "NO_REQUIREMENTS" }, 422);
    // Anthropic SDK errors carry a status; 5xx/timeout = provider outage → 503
    const status = (e as { status?: number }).status;
    if (typeof status === "number" && status >= 500) return err({ error: "SERVICE_UNAVAILABLE" }, 503);
    if ((e as Error).name === "APIConnectionTimeoutError") return err({ error: "SERVICE_UNAVAILABLE" }, 503);
    console.error("[gapmap] analyze failed:", (e as Error).name, (e as Error).message);
    return err({ error: "ANALYSIS_FAILED" }, 500);
  }
}
```

```ts
// lib/storage.ts
// All persistence is the browser's localStorage — GapMap stores nothing
// resume-derived on any server. Every value is wrapped in { v, data } so a
// future schema change can migrate instead of crash, and every read is
// defensive: localStorage is user-editable, i.e. untrusted input.

import type { AnalysisResult } from "@/types/analysis";
import { normalize } from "@/lib/normalize";

const INDEX_KEY = "gapmap:history:index";
const VERSION = 1;

export interface StoredAnalysis {
  result: AnalysisResult;
  resumeText: string;
  jdText: string;
  jdHash: string; // SHA-256 of normalized JD — lets History group runs of the same JD
}

function read<T>(key: string): T | null {
  try {
    const raw = localStorage.getItem(key);
    if (!raw) return null;
    const parsed = JSON.parse(raw) as { v: number; data: T };
    return parsed.v === VERSION ? parsed.data : null; // unknown version = treat as absent
  } catch { return null; }
}
function write(key: string, data: unknown): void {
  localStorage.setItem(key, JSON.stringify({ v: VERSION, data }));
}

function readIndex(): string[] { return read<string[]>(INDEX_KEY) ?? []; }

export function saveAnalysis(entry: StoredAnalysis): void {
  const attempt = () => {
    write(`gapmap:analysis:${entry.result.id}`, entry);
    write(INDEX_KEY, [entry.result.id, ...readIndex().filter((id) => id !== entry.result.id)]);
  };
  try { attempt(); } catch {
    // QuotaExceeded: evict the oldest half of history, retry once. If it
    // still fails, let it throw — the caller shows a message (never silent).
    const ids = readIndex();
    ids.slice(Math.ceil(ids.length / 2)).forEach(deleteAnalysis);
    attempt();
  }
}

export function loadAnalysis(id: string): StoredAnalysis | null {
  return read<StoredAnalysis>(`gapmap:analysis:${id}`);
}

export function listAnalyses(): StoredAnalysis[] {
  return readIndex().map(loadAnalysis).filter((e): e is StoredAnalysis => e !== null);
}

export function deleteAnalysis(id: string): void {
  localStorage.removeItem(`gapmap:analysis:${id}`);
  write(INDEX_KEY, readIndex().filter((x) => x !== id));
}

export function saveDraft(field: "resume" | "jd", text: string): void {
  try { write(`gapmap:draft:${field}`, text); } catch { /* drafts are best-effort */ }
}
export function loadDraft(field: "resume" | "jd"): string {
  return read<string>(`gapmap:draft:${field}`) ?? "";
}

export async function hashJd(jdText: string): Promise<string> {
  const bytes = new TextEncoder().encode(normalize(jdText));
  const digest = await crypto.subtle.digest("SHA-256", bytes);
  return Array.from(new Uint8Array(digest)).map((b) => b.toString(16).padStart(2, "0")).join("");
}
```

```ts
// lib/useShareReady.ts
// One async resource (fully-rendered card incl. fonts) gates BOTH download
// and copy. A single shared readiness check means the two buttons can never
// disagree about whether the card is exportable.
import { useEffect, useState, type RefObject } from "react";

export function useShareReady(cardRef: RefObject<HTMLElement | null>): boolean {
  const [ready, setReady] = useState(false);
  useEffect(() => {
    let cancelled = false;
    document.fonts.ready.then(() => { if (!cancelled && cardRef.current) setReady(true); });
    return () => { cancelled = true; };
  }, [cardRef]);
  return ready;
}
```

```ts
// lib/pdf.ts
// Client-side PDF text extraction — the file NEVER leaves the browser.
export async function extractPdfText(file: File): Promise<{ text: string; garbled: boolean }> {
  const pdfjs = await import("pdfjs-dist");
  pdfjs.GlobalWorkerOptions.workerSrc =
    new URL("pdfjs-dist/build/pdf.worker.min.mjs", import.meta.url).toString();
  const doc = await pdfjs.getDocument({ data: await file.arrayBuffer() }).promise;
  const pages: string[] = [];
  for (let p = 1; p <= doc.numPages; p++) {
    const content = await (await doc.getPage(p)).getTextContent();
    pages.push(content.items.map((it) => ("str" in it ? it.str : "")).join(" "));
  }
  const text = pages.join("\n\n").replace(/[ \t]+/g, " ").trim();
  // Garble heuristic: real resumes are mostly letters. Below 60% letterish
  // chars (or empty) the extraction probably mangled columns/ligatures.
  const letterish = (text.match(/[a-zA-Z0-9\s.,;:'"()\-\/&@+#]/g) ?? []).length;
  return { text, garbled: text.length === 0 || letterish / text.length < 0.6 };
}
```

Errors (encrypted/corrupt PDF) → visible message, textarea untouched.
2. Truncate extracted text at `RESUME_MAX_CHARS` with a notice if longer.

```ts
// eval/run-eval.ts
// The eval harness: runs the REAL pipeline over the gold set and reports
// extraction coverage, verdict accuracy (with a confusion matrix), and the
// fabrication-catch rate. Run with tsx: npm run eval -- --judge-model=...
// Costs real API money (~$0.10-0.30/run) — deliberate, not CI-automatic.

import { readdirSync, readFileSync, writeFileSync } from "node:fs";
import path from "node:path";
import { runAnalysis } from "@/lib/pipeline/pipeline";
import { getAnthropicClient } from "@/lib/anthropic";
import { JUDGE_MODEL, RESUME_MAX_CHARS } from "@/lib/constants";
import type { VerdictLabel, ScoreBreakdown } from "@/types/analysis";

interface GoldExpected { keywords: string[]; expectedVerdict: VerdictLabel; note?: string }
interface GoldPair {
  name: string; resumeText: string; jdText: string;
  expected: GoldExpected[]; expectedBandOneOf: ScoreBreakdown["band"][];
}

const judgeModel = process.argv.find((a) => a.startsWith("--judge-model="))?.split("=")[1] ?? JUDGE_MODEL;
const goldDir = path.join(process.cwd(), "eval", "golden");

function loadGold(): GoldPair[] {
  return readdirSync(goldDir).filter((f) => f.endsWith(".json")).sort().map((f) => {
    const pair = JSON.parse(readFileSync(path.join(goldDir, f), "utf8")) as GoldPair;
    // Self-validate: a malformed gold file must fail loudly, not skew results.
    if (!pair.name || !pair.resumeText || !pair.jdText || !Array.isArray(pair.expected) || !Array.isArray(pair.expectedBandOneOf)) {
      throw new Error(`malformed gold file: ${f}`);
    }
    return pair;
  });
}

async function main() {
  const client = getAnthropicClient();
  const gold = loadGold();
  console.log(`GapMap eval — judge model: ${judgeModel} — ${gold.length} pairs\n`);

  // 3x3 confusion matrix: confusion[expected][actual]
  const labels: VerdictLabel[] = ["match", "partial", "missing"];
  const confusion = Object.fromEntries(labels.map((e) => [e, { match: 0, partial: 0, missing: 0 }])) as
    Record<VerdictLabel, Record<VerdictLabel, number>>;
  let expectedTotal = 0, covered = 0, correct = 0;
  let quotesTotal = 0, quotesDropped = 0;
  const bandFailures: string[] = [];
  const perPair: unknown[] = [];

  for (const pair of gold) {
    const result = await runAnalysis(client, judgeModel, { resumeText: pair.resumeText, jdText: pair.jdText });
    const reqText = new Map(result.extraction.requirements.map((r) => [r.id, r.text.toLowerCase()]));

    let pairCovered = 0, pairCorrect = 0;
    for (const exp of pair.expected) {
      expectedTotal++;
      // Alignment: expected entry ↔ first extracted requirement containing all keywords
      const match = result.extraction.requirements.find((r) =>
        exp.keywords.every((k) => (reqText.get(r.id) ?? "").includes(k)));
      if (!match) continue;
      covered++; pairCovered++;
      const actual = result.judgments.find((j) => j.requirementId === match.id)!.verdict;
      confusion[exp.expectedVerdict][actual]++;
      if (actual === exp.expectedVerdict) { correct++; pairCorrect++; }
    }
    for (const j of result.judgments) {
      quotesDropped += j.droppedQuoteCount;
      quotesTotal += j.evidence.length + j.droppedQuoteCount;
    }
    const bandOk = pair.expectedBandOneOf.includes(result.scoreBreakdown.band);
    if (!bandOk) bandFailures.push(`${pair.name}: got ${result.scoreBreakdown.band} (${result.scoreBreakdown.score}), wanted ${pair.expectedBandOneOf.join("|")}`);
    perPair.push({ pair: pair.name, score: result.scoreBreakdown.score, band: result.scoreBreakdown.band,
                   bandOk, covered: pairCovered, correct: pairCorrect, expected: pair.expected.length });
    console.log(`  ${pair.name}: score=${result.scoreBreakdown.score} band=${result.scoreBreakdown.band}${bandOk ? "" : "  ⚠ BAND"} verdicts ${pairCorrect}/${pairCovered} (of ${pair.expected.length} expected)`);
  }

  const coverage = ((100 * covered) / expectedTotal).toFixed(1);
  const accuracy = covered ? ((100 * correct) / covered).toFixed(1) : "0";
  const fabRate = quotesTotal ? ((100 * quotesDropped) / quotesTotal).toFixed(1) : "0";
  console.log(`\nExtraction coverage: ${coverage}% (${covered}/${expectedTotal})`);
  console.log(`Verdict accuracy:    ${accuracy}% (${correct}/${covered})`);
  console.log(`Confusion (expected → actual):`, JSON.stringify(confusion));
  console.log(`Quote fabrication caught: ${fabRate}% (${quotesDropped}/${quotesTotal} quotes dropped)`);
  console.log(bandFailures.length ? `BAND FAILURES:\n  ${bandFailures.join("\n  ")}` : "All score bands sane.");

  writeFileSync(path.join(process.cwd(), "eval", `last-run-${judgeModel}.json`),
    JSON.stringify({ judgeModel, ranAt: new Date().toISOString(), coverage, accuracy, confusion, fabRate, bandFailures, perPair }, null, 2));
}

main().catch((e) => { console.error(e); process.exit(1); });
```

Note: `tsx` must resolve the `@/` alias — add `"tsconfig-paths"` behavior by running with `tsx --tsconfig tsconfig.json`; if `@/` fails under tsx, switch the script's imports to relative paths (`../lib/...`) — do NOT duplicate any logic.
